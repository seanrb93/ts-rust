# Virus, from the later games: a free for all in which one player starts infected and whoever an infected player touches is infected too.
# The last survivor wins.  The game of it is the script scripts/virus.rhai; this file makes the mode, sets its options and leaves out the ones of the Options page that mean nothing for it.
mode deathmatch
  variant "Virus"
  script virus.rhai
  team_play off
  score_limit 0
  kill_limit 0
  time_limit 0
  hide team_play team_starts friendly_fire kill_limit score_method score_limit one_shot_kill
